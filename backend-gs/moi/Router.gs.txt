// Phân quyền theo TOKEN (xem tokInfo trong Helpers.gs), không theo tham số.
function _chiAdmin(e) { var ti = tokInfo(e); return !!(ti && ti.role === 'admin'); }
function _laDuAn(e) { var ti = tokInfo(e); return !!(ti && ti.role === 'duan'); }
function _khongPhaiKhach(e) { var ti = tokInfo(e); return !!(ti && ti.role !== 'khachhang'); }
function doGet(e) {
  var action = e.parameter.action || '';
  var cb     = e.parameter.callback || '';
    var APP_KEY='QDy3MCObF1FLdoBexB9XO3dECy64';
  if((e.parameter.k||'')!==APP_KEY){
    var errPayload=JSON.stringify({status:'error', msg:'unauthorized'});
    return ContentService.createTextOutput(cb? (cb+'('+errPayload+')') : errPayload)
      .setMimeType(cb?ContentService.MimeType.JAVASCRIPT:ContentService.MimeType.JSON);
  }
  var CACHEABLE = {getGia:1,getImages:1,getSpExtra:1,getTonKho:1,getThuocTinhSP:1,getTonKhoChiTiet:1,getTrongLuong:1,getCT:1};
  var cache = CacheService.getScriptCache();
  var cacheKey = CACHEABLE[action] ? ('c_'+action+'_'+(e.parameter.loai||'')) : null;
  var result;
  if(cacheKey){
      result = _cacheGet(cache, cacheKey);
  }
  if(!result){
    try {
      if      (action === 'getGia')      result = getGia(e);
      else if (action === 'getGiaNgoiKeoKinh') result = getGiaNgoiKeoKinh();
      else if (action === 'getCT')       result = getCT(e);
      else if (action === 'logCT3Request') result = logCT3Request(e);
      else if (action === 'getImages')   result = getImages();
      else if (action === 'luuDon')      result = luuDon(e);
      else if (action === 'getTonKho')   result = getTonKho(e);
      else if (action === 'login')       result = login(e);
      else if (action === 'themUser')    result = _chiAdmin(e) ? themUser(e) : {status:'error', msg:'Chỉ admin mới được tạo/sửa tài khoản'};
      else if (action === 'getDonHang')  result = getDonHang(e);
      else if (action === 'getKhachHang')result = _khongPhaiKhach(e) ? getKhachHang() : {status:'error', msg:'forbidden'};
      else if (action === 'listTabs')    result = _chiAdmin(e) ? listTabs() : {status:'error', msg:'forbidden'};
      else if (action === 'getTonKhoChiTiet') result = getTonKhoChiTiet();
      else if (action === 'getTrongLuong')    result = getTrongLuong();
      else if (action === 'getThuocTinhSP')   result = getThuocTinhSP();
      else if (action === 'getAll')           result = getAll();
      else if (action === 'getSpExtra') result = getSpExtra();
            else if (action === 'getGiaDuAn') result = getGiaDuAn(e);
      else if (action === 'getHangOm')           result = _laNhanVien(e) ? getHangOm(e) : {status:'error', msg:'forbidden'};
      else if (action === 'luuHangOm')           result = _laNhanVien(e) ? luuHangOm(e) : {status:'error', msg:'forbidden'};
      else if (action === 'doiTrangThaiHangOm')  result = _laNhanVien(e) ? doiTrangThaiHangOm(e) : {status:'error', msg:'forbidden'};
      else if (action === 'getDanhMucDaiLy')     result = _laNhanVien(e) ? getDanhMucDaiLy() : {status:'error', msg:'forbidden'};
      else if (action === 'getGiaTBVS') result = getGiaTBVS();
      else if (action === 'clearCache') result = clearCache();
      else result = {status:'error', msg:'Unknown action: ' + action};
    } catch(err) {
      result = {status:'error', msg: err.toString()};
    }
    if(cacheKey && result && result.status==='ok'){
    _cacheSet(cache, cacheKey, result);
    }
  }
    if(action==='login' && result && result.status==='ok'){
    result.token = Utilities.getUuid();
    // Lưu "role|username": server tự biết ai đang gọi từ token, không tin
    // role/username do trình duyệt gửi lên.
    var _tokVal = String(result.role||'nv') + '|' + String(e.parameter.username||'').trim().toLowerCase();
    CacheService.getScriptCache().put('tok_'+result.token, _tokVal, 21600);
    PropertiesService.getScriptProperties().setProperty('tok_'+result.token, _tokVal);
  }
  if(action==='login' && result && result.status==='ok' && (result.role==='admin'||result.role==='duan')){ result.duanLabel = DUAN_NHAN; if(result.role==='duan') result.rieng = 1; }
  var _t = e.parameter.t || '';
  var _tokOK = _t && (CacheService.getScriptCache().get('tok_'+_t) || PropertiesService.getScriptProperties().getProperty('tok_'+_t));
  function _anGiaDL(d){
    d.nhan=0; d.giao=0; d.ns=0; d.gs=0; d.ct150nk=0; d.ct150gh=0;
    d.nhan2=0; d.giao2=0; d.nhan3=0; d.giao3=0;
  }
  var _laDA = _laDuAn(e) || (_chiAdmin(e) && e.parameter.v === 'da');
  if(!_tokOK || _laDA){
    if(action==='getGia' && result && result.data){
      result.data.forEach(_anGiaDL);
    }
    if(action==='getGiaNgoiKeoKinh' && result){
      ['ngoi','keo','kinh'].forEach(function(k){
        if(result[k] && result[k].data) result[k].data.forEach(_anGiaDL);
      });
    }
        if(action==='getGiaTBVS' && result && result.data){
      result.data.forEach(_anGiaDL);
    }
    if(action==='getCT'||action==='getKhachHang'||action==='themUser'||(!_tokOK&&(action==='getDonHang'||action==='getTonKhoChiTiet'||action==='getTonKho'))){
      result = {status:'error', msg:'need_login'};
    }
    if(action==='getAll' && result){
      if(result.gach && result.gach.data){
        result.gach.data.forEach(_anGiaDL);
            if(result.tbvs && result.tbvs.data){
        result.tbvs.data.forEach(_anGiaDL);
      }}
      if(result.ngoiKeoKinh){
        ['ngoi','keo','kinh'].forEach(function(k){
          if(result.ngoiKeoKinh[k] && result.ngoiKeoKinh[k].data) result.ngoiKeoKinh[k].data.forEach(_anGiaDL);
        });
      }
      result.ct1 = {status:'error', msg:'need_login'};
      result.ct2 = {status:'error', msg:'need_login'};
    }
  }
  if(!_laDA && result && result.status==='ok' && (action==='getGia'||action==='getAll')){
    var _an = _maAnKhoiDaiLy();
    if(_an.length){
      if(action==='getGia' && result.data) result.data = _loaiMaAn(result.data, _an);
      else if(action==='getAll' && result.gach && result.gach.data) result.gach.data = _loaiMaAn(result.gach.data, _an);
    }
  }
  if(_laDA && result && result.status==='ok' && (action==='getGia'||action==='getAll')){
    var _gd = _gachDuAn(e, action==='getGia' ? result.data : (result.gach && result.gach.data));
    if(_gd){
      if(action==='getGia') result.data = _gd;
      else if(result.gach) result.gach.data = _gd;
    }
  }
  // Khách hàng: KHÔNG gửi số tồn thật xuống trình duyệt. Mọi số lượng bị chặn
  // trần 51 (app hiển thị "50+ thùng" khi > 50), nên khách có mở DevTools
  // cũng chỉ thấy 51. Chạy SAU bước cache nên cache vẫn giữ số thật cho nhân viên.
  var _ti = tokInfo(e);
  if(_ti && (_ti.role==='khachhang'||_ti.role==='duan') && result && result.status==='ok'){
    var _cap = function(n){ n = Number(n)||0; return n>51 ? 51 : n; };
    if(action==='getTonKhoChiTiet' && result.data){
      Object.keys(result.data).forEach(function(ma){
        var d = result.data[ma]; if(!d) return;
        d.tong = _cap(d.tong);
        if(d.tier){ ['nhanh','mai','cho15'].forEach(function(t){ d.tier[t] = _cap(d.tier[t]); }); }
        (d.kho||[]).forEach(function(k){
          k.tong = _cap(k.tong);
          (k.lo||[]).forEach(function(l){ l.sl = _cap(l.sl); });
        });
      });
    }
    if(action==='getTonKho' && result.data){
      Object.keys(result.data).forEach(function(ma){
        var d = result.data[ma]; if(!d) return;
        d.kha_dung = _cap(d.kha_dung); d.co_san = _cap(d.co_san); d.vien_kd = 0;
      });
    }
  }
  var json = JSON.stringify(result);
  var out  = cb ? cb + '(' + json + ')' : json;
  var mime = cb ? ContentService.MimeType.JAVASCRIPT : ContentService.MimeType.JSON;
  return ContentService.createTextOutput(out).setMimeType(mime);
}
